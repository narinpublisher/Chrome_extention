7-1   Handling Clicks Efficiently: The Power of Event Delegation

📄  script.js (build the full task row)
---------------------------------------------
function addTask(taskText) {
  const li = document.createElement('li');
 
  const checkbox = document.createElement('input');
  checkbox.type = 'checkbox';
  checkbox.className = 'task-checkbox';
 
  const span = document.createElement('span');
  span.textContent = taskText;
  span.className = 'task-text';
 
  const deleteBtn = document.createElement('button');
  deleteBtn.type = 'button';
  deleteBtn.textContent = '✕';
  deleteBtn.className = 'delete-btn';
 
  li.append(checkbox, span, deleteBtn);
  list.appendChild(li);
}
---------------------------------------------


📄  script.js (event delegation for checking off and deleting)
-----------------------------------------------------------------------------
list.addEventListener('click', (event) => {
  const li = event.target.closest('li');
  if (!li) return;
 
  if (event.target.classList.contains('task-checkbox')) {
    li.classList.toggle('completed');
  }
 
  if (event.target.classList.contains('delete-btn')) {
    li.remove();
  }
});

function addTask(taskText, id = Date.now(), completed = false) {
  const li = document.createElement('li');

  const checkbox = document.createElement('input');
  checkbox.type = 'checkbox';
  checkbox.className = 'task-checkbox';
  checkbox.checked = completed;

  span.textContent = taskText;
  span.className = 'task-text';
 
  const deleteBtn = document.createElement('button');
  deleteBtn.type = 'button';
  deleteBtn.textContent = '✕';
  deleteBtn.className = 'delete-btn';
 
  li.append(checkbox, span, deleteBtn);
  list.appendChild(li);
}

7-3   Saving and Restoring Tasks with chrome.storage.sync

📄  script.js (saveTasks)
-------------------------------
function saveTasks() {
  const tasks = [...list.children].map((li) => ({
    id: Number(li.dataset.id),
    text: li.querySelector('.task-text').textContent,
    completed: li.classList.contains('completed'),
  }));
 
  chrome.storage.sync.set({ tasks });
}
-------------------------------

📄  script.js (saving on every change)

form.addEventListener('submit', (event) => {

7-4   Multi-Window Syncing: Listening for Storage Changes

📄  script.js (renderTasks, load, and live sync)
------------------------------------------------------------
function renderTasks(tasks) {
  list.innerHTML = '';
  tasks.forEach((task) => {
    addTask(task.text, task.id, task.completed);
  });
}
 
chrome.storage.sync.get(['tasks'], (result) => {
  renderTasks(result.tasks || []);
});
 
chrome.storage.onChanged.addListener((changes, area) => {
  if (area === 'sync' && changes.tasks) {
    renderTasks(changes.tasks.newValue || []);
  }
});
------------------------------------------------------------
